        import React from 'react';
        import { BladeProvider } from '@greenloom/ui/components';
        import { bladeTheme } from '@greenloom/ui/tokens';

        export function App() {
          return (
            <BladeProvider themeTokens={bladeTheme} colorScheme="light">
              <YourApplicationRoutes />
            </BladeProvider>
          );
        }

    <Heading size="large" marginTop="spacing.7">2. Typography Configuration</Heading>

    <Text marginY="spacing.4" color="surface.text.gray.subtle">
      Loom UI standardizes on <strong>Geist</strong> for all headings, body text, and UI labels, and <strong>Geist Mono</strong> for numeric figures, currencies, tables, counters, and code.
    </Text>

    ### Google Fonts Import (Recommended for Web)

    Add the Google Fonts stylesheet to your `<head>` or main entry file:

    ```html
    <link rel="preconnect" href="https://fonts.googleapis.com" />
    <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin />
    <link
      href="https://fonts.googleapis.com/css2?family=Geist:wght@100..900&family=Geist+Mono:wght@100..900&display=swap"
      rel="stylesheet"
    />
    ```

    ### CSS / Global Styles

    ```css
    body {
      font-family: 'Geist', -apple-system, BlinkMacSystemFont, "Segoe UI", Roboto, sans-serif;
      -webkit-font-smoothing: antialiased;
    }

    code, kbd, samp, pre, .mono {
      font-family: 'Geist Mono', ui-monospace, SFMono-Regular, Menlo, monospace;
    }
    ```
